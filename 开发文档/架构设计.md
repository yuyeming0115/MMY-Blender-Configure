# MMY Blender Toolkit 架构设计

## 概述

### 设计目标

将 MMY Blender Configure 从单一功能插件升级为**可扩展的模块化工具集**，实现：

- **即插即用**：新功能作为独立模块添加，不影响现有模块
- **统一入口**：一个面板管理所有功能，避免 UI 分散
- **用户可控**：每个模块可独立启用/禁用
- **低耦合**：模块间无强制依赖，可单独开发测试

### 命名变更

| 原名称 | 新名称 |
|--------|--------|
| MMY Blender Configure | MMY Blender Toolkit |
| 分类：MMY-Tools | 保持不变 |

## 核心框架

### 模块基类 (`core/base.py`)

```python
from abc import ABC, abstractmethod
from typing import List, Dict, Any, Tuple, Optional
import bpy.types


class MMYModuleBase(ABC):
    """
    模块基类，所有功能模块必须继承此类。

    子类必须实现:
        - register(): 注册 Operators、Panels、Handlers 等
        - unregister(): 注销所有注册的内容

    子类可选实现:
        - draw_menu(): 在主面板中绘制模块专属区域
        - draw_preferences(): 在偏好设置中绘制模块配置
        - on_enable(): 模块启用时的回调
        - on_disable(): 模块禁用时的回调
    """

    # ========== 元信息（子类必须定义）==========

    module_id: str                    # 唯一标识符，如 "config_backup"
    display_name: str                 # 显示名称，如 "配置备份"
    description: str                  # 模块描述
    version: Tuple[int, ...] = (1, 0, 0)
    author: str = ""                  # 模块作者
    blender_min: Tuple[int, ...] = (4, 5, 0)  # 最低 Blender 版本

    # ========== 依赖关系 ==========

    # 依赖的其他模块 ID 列表
    dependencies: List[str] = []

    # 冲突的模块 ID 列表（与这些模块不能同时启用）
    conflicts: List[str] = []

    # ========== 分类与排序 ==========

    # 模块分类（用于 UI 分组）
    category: str = "general"         # general, workflow, rendering, materials

    # 同分类内的排序权重（越小越靠前）
    priority: int = 100

    # ========== 注册与注销 ==========

    @abstractmethod
    def register(self) -> None:
        """
        注册模块的 Blender 类（Operators、Panels、Handlers 等）。

        注意:
            - 只注册模块专属的类，不要注册其他模块的类
            - 注册的类名建议以 MMY_OT_、MMY_PT_ 等前缀开头
        """
        pass

    @abstractmethod
    def unregister(self) -> None:
        """
        注销模块的所有 Blender 类。

        注意:
            - 必须按照注册的逆序注销
            - 确保所有注册的内容都被注销，避免内存泄漏
        """
        pass

    # ========== 可选回调 ==========

    def on_enable(self) -> None:
        """模块启用时调用（用户在偏好设置中开启模块）"""
        pass

    def on_disable(self) -> None:
        """模块禁用时调用（用户在偏好设置中关闭模块）"""
        pass

    # ========== UI 绘制 ==========

    def draw_menu(self, layout: bpy.types.UILayout, context: bpy.types.Context) -> None:
        """
        在主面板中绘制模块专属区域。

        Args:
            layout: Blender UI 布局对象
            context: Blender 上下文对象
        """
        pass

    def draw_preferences(self, layout: bpy.types.UILayout, context: bpy.types.Context) -> None:
        """
        在插件偏好设置中绘制模块专属配置项。

        Args:
            layout: Blender UI 布局对象
            context: Blender 上下文对象
        """
        pass

    # ========== 工具方法 ==========

    def get_preference(self, key: str, default: Any = None) -> Any:
        """获取模块偏好设置值"""
        from ..preferences import get_module_preference
        return get_module_preference(self.module_id, key, default)

    def set_preference(self, key: str, value: Any) -> None:
        """设置模块偏好设置值"""
        from ..preferences import set_module_preference
        set_module_preference(self.module_id, key, value)

    def is_enabled(self) -> bool:
        """检查模块是否已启用"""
        from ..preferences import is_module_enabled
        return is_module_enabled(self.module_id)
```

### 模块注册中心 (`core/__init__.py`)

```python
from typing import Dict, List, Type, Optional
from .base import MMYModuleBase


class ModuleRegistry:
    """
    模块注册中心，负责模块的注册、发现、依赖解析和生命周期管理。
    """

    # 已注册的模块字典
    _modules: Dict[str, MMYModuleBase] = {}

    # 模块加载顺序（考虑依赖关系）
    _load_order: List[str] = []

    # 是否已初始化
    _initialized: bool = False

    # ========== 注册与注销 ==========

    @classmethod
    def register_module(cls, module_class: Type[MMYModuleBase]) -> None:
        """
        注册一个模块。

        Args:
            module_class: 模块类（非实例）

        Raises:
            ValueError: 模块 ID 已存在或依赖未满足
        """
        module = module_class()
        module_id = module.module_id

        if module_id in cls._modules:
            raise ValueError(f"模块 '{module_id}' 已注册")

        # 检查依赖
        missing_deps = [dep for dep in module.dependencies if dep not in cls._modules]
        if missing_deps:
            raise ValueError(f"模块 '{module_id}' 缺少依赖: {missing_deps}")

        # 检查冲突
        conflicts = [c for c in module.conflicts if c in cls._modules]
        if conflicts:
            raise ValueError(f"模块 '{module_id}' 与以下模块冲突: {conflicts}")

        cls._modules[module_id] = module
        cls._update_load_order()

    @classmethod
    def unregister_all(cls) -> None:
        """注销所有模块（按加载逆序）"""
        for module_id in reversed(cls._load_order):
            module = cls._modules[module_id]
            if cls.is_module_active(module_id):
                module.unregister()
        cls._modules.clear()
        cls._load_order.clear()
        cls._initialized = False

    # ========== 模块激活与停用 ==========

    @classmethod
    def activate_module(cls, module_id: str) -> bool:
        """
        激活模块（调用 register）。

        Args:
            module_id: 模块 ID

        Returns:
            是否成功激活
        """
        module = cls._modules.get(module_id)
        if not module:
            return False

        if cls.is_module_active(module_id):
            return True  # 已激活

        # 检查依赖是否都已激活
        for dep in module.dependencies:
            if not cls.is_module_active(dep):
                return False

        try:
            module.register()
            module.on_enable()
            return True
        except Exception as e:
            print(f"[MMY Toolkit] 激活模块 '{module_id}' 失败: {e}")
            return False

    @classmethod
    def deactivate_module(cls, module_id: str) -> bool:
        """
        停用模块（调用 unregister）。

        Args:
            module_id: 模块 ID

        Returns:
            是否成功停用
        """
        module = cls._modules.get(module_id)
        if not module:
            return False

        if not cls.is_module_active(module_id):
            return True  # 已停用

        # 检查是否有其他模块依赖此模块
        dependents = cls.get_dependents(module_id)
        active_dependents = [d for d in dependents if cls.is_module_active(d)]
        if active_dependents:
            print(f"[MMY Toolkit] 无法停用模块 '{module_id}'，以下模块依赖它: {active_dependents}")
            return False

        try:
            module.on_disable()
            module.unregister()
            return True
        except Exception as e:
            print(f"[MMY Toolkit] 停用模块 '{module_id}' 失败: {e}")
            return False

    @classmethod
    def is_module_active(cls, module_id: str) -> bool:
        """检查模块是否已激活"""
        from ..preferences import is_module_enabled
        return is_module_enabled(module_id)

    # ========== 查询方法 ==========

    @classmethod
    def get_module(cls, module_id: str) -> Optional[MMYModuleBase]:
        """获取模块实例"""
        return cls._modules.get(module_id)

    @classmethod
    def get_all_modules(cls) -> List[MMYModuleBase]:
        """获取所有模块（按加载顺序）"""
        return [cls._modules[mid] for mid in cls._load_order]

    @classmethod
    def get_modules_by_category(cls) -> Dict[str, List[MMYModuleBase]]:
        """按分类获取模块"""
        categories: Dict[str, List[MMYModuleBase]] = {}
        for module in cls.get_all_modules():
            cat = module.category
            if cat not in categories:
                categories[cat] = []
            categories[cat].append(module)
        return categories

    @classmethod
    def get_dependents(cls, module_id: str) -> List[str]:
        """获取依赖指定模块的所有模块"""
        return [
            mid for mid, m in cls._modules.items()
            if module_id in m.dependencies
        ]

    # ========== 内部方法 ==========

    @classmethod
    def _update_load_order(cls) -> None:
        """更新模块加载顺序（拓扑排序，考虑依赖关系）"""
        # 拓扑排序实现
        visited = set()
        order = []

        def visit(module_id: str):
            if module_id in visited:
                return
            visited.add(module_id)
            module = cls._modules[module_id]
            for dep in module.dependencies:
                if dep in cls._modules:
                    visit(dep)
            order.append(module_id)

        # 按优先级排序后进行拓扑排序
        sorted_modules = sorted(
            cls._modules.values(),
            key=lambda m: m.priority
        )
        for module in sorted_modules:
            visit(module.module_id)

        cls._load_order = order


# 便捷导出
__all__ = ['ModuleRegistry', 'MMYModuleBase']
```

### 模块发现与加载 (`modules/__init__.py`)

```python
import importlib
import pkgutil
from pathlib import Path
from typing import List, Type

from ..core.base import MMYModuleBase
from ..core import ModuleRegistry


def discover_modules() -> List[Type[MMYModuleBase]]:
    """
    自动发现 modules 目录下的所有模块。

    发现规则:
        - 遍历 modules/ 下的所有子目录
        - 查找 __init__.py 中定义的 MODULE_CLASS 变量
        - 或查找继承自 MMYModuleBase 的类

    Returns:
        模块类列表
    """
    modules_dir = Path(__file__).parent
    discovered = []

    for item in modules_dir.iterdir():
        if item.is_dir() and not item.name.startswith('_'):
            module_name = f"{__package__}.{item.name}"
            try:
                module = importlib.import_module(module_name)

                # 方式1: 查找 MODULE_CLASS 变量
                if hasattr(module, 'MODULE_CLASS'):
                    discovered.append(module.MODULE_CLASS)

                # 方式2: 查找继承自 MMYModuleBase 的类
                else:
                    for attr_name in dir(module):
                        attr = getattr(module, attr_name)
                        if (isinstance(attr, type) and
                            issubclass(attr, MMYModuleBase) and
                            attr is not MMYModuleBase):
                            discovered.append(attr)
                            break

            except Exception as e:
                print(f"[MMY Toolkit] 发现模块 '{item.name}' 失败: {e}")

    return discovered


def load_all_modules() -> None:
    """发现并注册所有模块"""
    for module_class in discover_modules():
        try:
            ModuleRegistry.register_module(module_class)
        except ValueError as e:
            print(f"[MMY Toolkit] 注册模块失败: {e}")


__all__ = ['discover_modules', 'load_all_modules']
```

### 偏好设置管理 (`preferences.py`)

```python
import bpy
from bpy.props import BoolProperty, StringProperty, IntProperty, CollectionProperty
from bpy.types import PropertyGroup, AddonPreferences
from typing import Any, Dict, Optional

from .core import ModuleRegistry


class ModuleSettingItem(PropertyGroup):
    """单个模块的设置项（用于 CollectionProperty）"""
    module_id: StringProperty()
    enabled: BoolProperty(default=True)


class MMYToolkitPreferences(AddonPreferences):
    """插件偏好设置"""
    bl_idname = __package__

    # ========== 全局设置 ==========

    show_in_header: BoolProperty(
        name="在顶部栏显示",
        description="在 Blender 顶部菜单栏显示快速访问按钮",
        default=True
    )

    panel_width: IntProperty(
        name="面板宽度",
        description="弹出面板的宽度（像素）",
        default=300,
        min=200,
        max=600
    )

    # ========== 模块设置存储 ==========

    # 使用 CollectionProperty 存储各模块的启用状态
    module_settings: CollectionProperty(type=ModuleSettingItem)

    # ========== 内部方法 ==========

    def _get_module_setting(self, module_id: str) -> Optional[ModuleSettingItem]:
        """获取模块设置项"""
        for item in self.module_settings:
            if item.module_id == module_id:
                return item
        return None

    def _ensure_module_setting(self, module_id: str) -> ModuleSettingItem:
        """确保模块设置项存在"""
        item = self._get_module_setting(module_id)
        if item is None:
            item = self.module_settings.add()
            item.module_id = module_id
        return item

    # ========== UI 绘制 ==========

    def draw(self, context):
        layout = self.layout

        # 全局设置
        box = layout.box()
        box.label(text="全局设置", icon='PREFERENCES')
        box.prop(self, "show_in_header")
        box.prop(self, "panel_width")

        # 模块管理
        box = layout.box()
        box.label(text="模块管理", icon='PLUGIN')

        # 按分类显示模块
        categories = ModuleRegistry.get_modules_by_category()
        category_names = {
            'general': '通用',
            'workflow': '工作流',
            'rendering': '渲染',
            'materials': '材质'
        }

        for cat, modules in categories.items():
            cat_name = category_names.get(cat, cat)
            col = box.column(align=True)
            col.label(text=cat_name, icon='DOWNARROW_HLT')

            for module in modules:
                self._draw_module_setting(col, module)

    def _draw_module_setting(self, layout, module):
        """绘制单个模块的设置"""
        item = self._ensure_module_setting(module.module_id)

        row = layout.row(align=True)
        row.prop(item, 'enabled', text=module.display_name)

        # 模块专属设置按钮
        if hasattr(module, 'draw_preferences'):
            op = row.operator(
                "mmy.show_module_preferences",
                text="",
                icon='PREFERENCES'
            )
            op.module_id = module.module_id

        # 状态指示
        if item.enabled:
            # 检查依赖是否满足
            missing_deps = [
                dep for dep in module.dependencies
                if not is_module_enabled(dep)
            ]
            if missing_deps:
                row.label(text=f"⚠ 依赖未满足: {missing_deps}", icon='ERROR')


# ========== 全局辅助函数 ==========

def get_preferences() -> Optional[MMYToolkitPreferences]:
    """获取插件偏好设置实例"""
    import bpy
    return bpy.context.preferences.addons.get(__package__)


def is_module_enabled(module_id: str) -> bool:
    """检查模块是否启用"""
    prefs = get_preferences()
    if prefs is None:
        return True  # 默认启用

    item = prefs._get_module_setting(module_id)
    return item.enabled if item else True


def get_module_preference(module_id: str, key: str, default: Any = None) -> Any:
    """获取模块偏好设置值"""
    prefs = get_preferences()
    if prefs is None:
        return default

    item = prefs._get_module_setting(module_id)
    if item is None:
        return default

    return getattr(item, key, default)


def set_module_preference(module_id: str, key: str, value: Any) -> None:
    """设置模块偏好设置值"""
    prefs = get_preferences()
    if prefs is None:
        return

    item = prefs._ensure_module_setting(module_id)
    if hasattr(item, key):
        setattr(item, key, value)


# ========== 注册与注销 ==========

classes = [
    ModuleSettingItem,
    MMYToolkitPreferences,
]


def register():
    for cls in classes:
        bpy.utils.register_class(cls)


def unregister():
    for cls in reversed(classes):
        bpy.utils.unregister_class(cls)
```

```python
        for cat, modules in categories.items():
            # 分类标题
            icon = category_icons.get(cat, 'PLUGIN')
            box = layout.box()
            box.label(text=cat.upper(), icon=icon)

            # 绘制各模块
            for module in modules:
                if module.is_enabled():
                    self._draw_module(box, module, prefs)

    def _draw_module(self, layout, module, prefs):
        """绘制单个模块区域"""
        # 模块标题行
        row = layout.row(align=True)
        row.label(text=module.display_name, icon='PLUGIN')

        # 启用/禁用开关
        item = prefs._get_module_setting(module.module_id)
        if item:
            row.prop(item, 'enabled', text="")

        # 设置按钮
        if hasattr(module, 'draw_preferences'):
            op = row.operator(
                "mmy.show_module_preferences",
                text="",
                icon='PREFERENCES'
            )
            op.module_id = module.module_id

        # 模块专属内容
        col = layout.column(align=True)
        module.draw_menu(col, bpy.context)


class MMY_OT_show_main_panel(bpy.types.Operator):
```

```python
import bpy
import os


class MMY_OT_AddSuffix(bpy.types.Operator):
    """添加后缀到当前文件名"""
    bl_idname = "mmy.add_suffix"
    bl_label = "添加后缀"
    bl_options = {'REGISTER', 'UNDO'}

    suffix: bpy.props.StringProperty()

    def execute(self, context):
        # 获取文件浏览器的 active_operator
        for area in context.screen.areas:
            if area.type == 'FILE_BROWSER':
                space = area.spaces.active
                if space and hasattr(space, 'params'):
                    params = space.params
                    if params:
                        # 获取当前文件名
                        filename = params.filename
                        # 移除旧后缀（如果存在）
                        for old_suffix in ["_Mesh", "_Mat", "_Rig", "_Ani", "_Render"]:
                            if filename.endswith(old_suffix):
                                filename = filename[:-len(old_suffix)]
                                break
                        # 添加新后缀
                        base_name = os.path.splitext(filename)[0]
                        params.filename = base_name + self.suffix
                        break
        return {'FINISHED'}
```

## 设计决策

| 决策点 | 选择 | 理由 |
|--------|------|------|
| 模块发现机制 | 自动扫描目录 | 无需手动注册，新增模块只需放入目录 |
| 模块激活控制 | 偏好设置存储 | 用户可独立启用/禁用各模块 |
| 依赖解析 | 拓扑排序 | 确保模块按正确顺序加载 |
| 面板整合 | 单一主面板 + 模块子区域 | 避免 UI 分散，统一入口 |
| 命名空间 | 所有类名以 MMY_ 开头 | 避免与其他插件冲突 |
